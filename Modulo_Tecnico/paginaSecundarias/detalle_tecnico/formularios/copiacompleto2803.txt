import React, { useState, useEffect } from 'react';
import { useParams } from 'react-router-dom';

const CierreCasoCompletadoModal = (id_historial_cambioestado, id_datosvisita, estado_solicitud, onClose) => {
    const [comentarioTrabajo, setComentarioTrabajo] = useState('');
    const [tipoSolucionFalla, setTipoSolucionFalla] = useState('');
    const [solicitudAsigTec, setSolicitudAsigTec] = useState({
        historial: [],
        datosVisita:{},
        solicitud: {}
    });

    const [error, setError] = useState(null);
    const [loading, setLoading] = useState(true);
    
    const { codigo_solicitud } = useParams(); // Asegúrate de que id esté definido en la URL

    useEffect(() => {
        const fetchSolicitudAsig = async () => {
            try {
                const response = await fetch(`${process.env.REACT_APP_API_URL_SOLIASIGTEC}/api/datosformulariocompleto/${codigo_solicitud}`);
                
                if (!response.ok) {
                    throw new Error(`Error: ${response.status} ${response.statusText}`);
                }

                const data = await response.json();

                console.log('Datos recibidos de la API:', data);

                if(data){
                    setSolicitudAsigTec({
                        historial: data.historial || [],
                        datosVisita: data.datosVisita || {},
                        solicitud: data.solicitud || {}
                    });
                }else{
                    console.error('No se recibieron datos validos de la API');
                }

            } catch (error) {
                console.error('Error fetching solicitud:', error);
                setError(error.message);
            } finally {
                setLoading(false);
            }
        };

        fetchSolicitudAsig();
    }, [codigo_solicitud]);

    

    const handleSubmit = async (event) => {
        event.preventDefault();

        const datos = {
            comentario_trabajo_realizado: comentarioTrabajo,
            tipo_solucion_falla: tipoSolucionFalla,
            // Datos de solicitud
            codigo_solicitud: solicitudAsigTec.solicitud?.codigo_solicitud,
            estado_solicitud: solicitudAsigTec.solicitud?.estado_solicitud,
            // Datos de visita
            id_datosvisita: solicitudAsigTec.datosVisita?.id_datosvisita,
            // Historial
            id_historial_cambioestado: solicitudAsigTec.historial?.id_historial_cambioestado,
        };

        console.log('datos enviados:', datos);

        try {
            const response = await fetch(`${process.env.REACT_APP_API_URL_SOLIASIGTEC}/api/guardarsolicitudcompleta`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify(datos),
            });

            if (response.ok) {
                alert('Caso cerrado exitosamente');
                onClose(); // Cerrar el modal
            } else {
                alert('Error al cerrar el caso');
            }
            const result = await response.json();
                console.log('Resultado:', result);

            console.log('Datos enviados:', {
                codigo_solicitud, 
                estado_solicitud, 
                id_historial_cambioestado, 
                id_datosvisita, 
                tipoSolucionFalla, 
                comentarioTrabajo
                });
        } catch (error) {
            console.error('Error:', error);
            setError({ submit: 'Error asignando tu ticket' });
        }
    };

    if (error) {
        return <div>Error al cargar la solicitud: {error}</div>;
    }

    if (loading) {
        return <div>Cargando...</div>;
    }

    return (
        <div className="modal">
            <h2>Cierre Caso Completado</h2>
            <form onSubmit={handleSubmit}>
                <label>Solicitud: {solicitudAsigTec.solicitud?.codigo_solicitud || 'Sin codigo'}</label>
                <label htmlFor="tipoSolucionFalla">Solución/Falla:</label>
                <select
                    id="tipoSolucionFalla"
                    value={tipoSolucionFalla}
                    onChange={(e) => setTipoSolucionFalla(e.target.value)}
                    required
                >
                    <option value="">Seleccione una opción</option>
                    <option value="Config. Equipos">Config. Equipos</option>
                    <option value="Certificación">Certificación</option>
                    <option value="Cambio ONU">Cambio ONU</option>
                    <option value="Cambio Patch Card">Cambio Patch Card</option>
                    <option value="Cambio Conectores">Cambio Conectores</option>
                </select>
                
                <br />
                <label htmlFor="comentarioTrabajo">Comentario del Trabajo realizado:</label>
                <textarea
                    id="comentarioTrabajo"
                    value={comentarioTrabajo}
                    onChange={(e) => setComentarioTrabajo(e.target.value)}
                    required
                />
                <br />
                <input type="hidden" value={estado_solicitud} />
                <button type="submit">Guardar Cambios</button>
                <button type="button" onClick={onClose}>Cancelar</button>
            </form>
        </div>
    );
};

export default CierreCasoCompletadoModal;
