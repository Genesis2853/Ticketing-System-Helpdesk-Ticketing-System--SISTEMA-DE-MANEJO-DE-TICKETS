// server.js
import express from 'express';
import cors from 'cors';
import config from '../../config.js';
import dotenv from 'dotenv';
import apiEstadistico from './conexionpython.js'; // Importa el módulo src/python/conexionpython.js
import { autenticarToken, autorizarRoles, autorizarPorPermiso } from '../../Modulo_Usuario/modulo_usuario.js';

dotenv.config();

const app = express();
const port = config.portDatosEstadisticos || 8000;

app.use(cors());
app.use(express.json());

// Rutas para consumir las APIs de FastAPI usando el módulo
app.get('/api/solicitudes/cerradas/completadas', autenticarToken, autorizarRoles('Admin', 'Moderador'), autorizarPorPermiso('datos_estadisticos'), async (req, res) => {
    console.log("Token de acceso:", req.headers.authorization);
    try {
        const data = await apiEstadistico.solicitudesCerradasCompletadas();
        res.json(data);
    } catch (error) {
        console.error("Error en la solicitud:", error.message);
        res.status(500).json({ mensaje: "Error al obtener datos", error: error.message });
    }
});

// Agrega más rutas según sea necesario
app.get('/api/tickets/sin_asignar', autenticarToken, autorizarRoles('Admin', 'Moderador'), autorizarPorPermiso('datos_estadisticos'), async (req, res) => {
    try {
        const data = await apiEstadistico.ticketsSinAsignar();
        res.json(data);
    } catch (error) {
        console.error("Error en la solicitud:", error.message);
        res.status(500).json({ mensaje: "Error al obtener datos", error: error.message });
    }
});

// Puedes agregar más rutas aquí usando las funciones de apiEstadistico

app.listen(port, () => {
    console.log(`Servidor corriendo en http://localhost:${port}`);
});
