import React from "react";
import './estilodesempeño.css';

const TechnicianPerformance = ({ technicians, evaluaciones }) => {
  // Función para calcular el desempeño de los técnicos
  const calculatePerformance = (techniciansList, evaluationsList) => {
    return techniciansList.map(tech => {
      const techEvals = evaluationsList.filter(ev => ev.codigo_trabajador === tech.codigo_trabajador);
      const solicitudesCount = techEvals.length;
      const avg = solicitudesCount > 0 
        ? (techEvals.reduce((acc, ev) => acc + ev.puntuacion_tecnico, 0) / solicitudesCount) 
        : 0;

      return {
        ...tech,
        solicitudes: solicitudesCount,
        promedio: avg.toFixed(2),
      };
    });
  };

  const sortedTechnicians = calculatePerformance(technicians, evaluaciones).sort((a, b) => b.promedio - a.promedio);

  // Función para asignar categorías y estilos
  const getCategory = (promedio) => {
    if (promedio >= 4.5) return { label: "Excelente", color: "#57D131", icon: "⭐⭐⭐⭐⭐" };
    if (promedio >= 3.5) return { label: "Muy Bueno", color: "#A7F432", icon: "⭐⭐⭐⭐" };
    if (promedio >= 2.5) return { label: "Bueno", color: "#3F91F4", icon: "⭐⭐⭐" };
    if (promedio >= 1.5) return { label: "Regular", color: "#F4D03F", icon: "⭐⭐" };
    return { label: "Crítico", color: "#F44336", icon: "⭐" };
  };

  return (
    <div className="technician-performance">
      <h2>Desempeño de Técnicos</h2>
      <table>
        <thead>
          <tr>
            <th>Posición</th>
            <th>Técnico</th>
            <th>Solicitudes Evaluadas</th>
            <th>Puntuación</th>
            <th>Categoría</th>
          </tr>
        </thead>
        <tbody>
          {sortedTechnicians.map((tech, index) => {
            const { label, color, icon } = getCategory(tech.promedio);
            
            return (
              <tr key={tech.codigo_trabajador} className={index === 0 ? "first-place" : ""}>
                <td>{index === 0 ? "🏆" : index + 1}</td>
                <td>{tech.nombre_tecnico || "Nombre no disponible"}</td>
                <td>{tech.solicitudes}</td>
                <td>
                  <div className="stars">{icon}</div>
                  <div className="score">{isNaN(tech.promedio) ? "N/A" : Number(tech.promedio).toFixed(2)}</div>
                </td>
                <td style={{ backgroundColor: color }}>{label}</td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
};

export default TechnicianPerformance;
