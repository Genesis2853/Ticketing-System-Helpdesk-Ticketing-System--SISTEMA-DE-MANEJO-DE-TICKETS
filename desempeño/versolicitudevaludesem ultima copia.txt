import React, { useState, useEffect } from "react";

const RequestViewer = ({ solicitudes,  feedbacks }) => {
  const [evaluadas, setEvaluadas] = useState([]);

useEffect(() => {
        const fetchEvaluaciones = async () => {
            try {
                const response = await fetch(`${process.env.REACT_APP_API_URL_RETROCLIENTE}/api/evaluadas`);
                if (!response.ok) {
                    throw new Error('Error al obtener evaluaciones sin feedback');
                }
                const data = await response.json();
                setEvaluadas(data);
            } catch (error) {
                console.error('Error al obtener evaluaciones sin feedback:', error);
            }
        };

        fetchEvaluaciones();
    }, []); // Se ejecuta solo una vez al montar el componente



  // Filtrar solicitudes no evaluadas
  const solicitudesPendientes = solicitudes.filter(solicitud =>
    !evaluadas.some(evaluada => evaluada.id_soli_completada === solicitud.id_soli_completada)
  );

  return (
    <div className="request-viewer">
      <h2>Solicitudes</h2>
      
      {/* Sección de solicitudes pendientes */}
      <h3>Pendientes</h3>
      {solicitudesPendientes.length === 0 ? (
        <p>No hay solicitudes pendientes.</p>
      ) : (
        <div className="pending-requests">
          {solicitudesPendientes.map((solicitud) => {
            if (!solicitud || !solicitud.id_soli_completada) {
              return null; // No renderizar si no hay id_soli_completada
            }
            return (
              <div key={solicitud.id_soli_completada} className="request-item">
                <p><strong>{solicitud.id_soli_completada}</strong></p>
                - Técnico: {solicitud.nombre_tecnico} {solicitud.apellido_tecnico}
              </div>
            );
          })}
        </div>
      )}

      {/* Sección de solicitudes evaluadas */}
      <h3>✅ Evaluadas</h3>
      {evaluadas.length === 0 ? (
        <p>No hay evaluaciones registradas.</p>
      ) : (
        <div className="evaluated-requests">
          {evaluadas.map((evaluacion) => {
            // Verificar si hay feedback del cliente
            // Convertir ambos ids a número para comparación segura
            const evaluacionIdFeedback = evaluacion.id_feedback === null ? null : Number(evaluacion.id_feedback);
            const tieneFeedback = feedbacks.some(fb => Number(fb.id_feedback) === evaluacionIdFeedback);

            return (
              <div 
                key={evaluacion.id_evaluaciones} // Asegúrate de que este ID sea único para cada evaluación
                className="request-item" 
                style={{ 
                  display: 'flex', 
                  justifyContent: 'space-between', 
                  alignItems: 'center',
                  gap: '10px'
                }}
              >
                <div>
                  <strong>{evaluacion.id_soli_completada}</strong> - Técnico: {evaluacion.nombre_tecnico} {evaluacion.apellido_tecnico} 
                  <p><strong>Puntuación Técnica:</strong> {evaluacion.puntuacion_tecnico}</p>
                  <p><strong>Comentario Técnico:</strong> {evaluacion.comentario_puntuacion_tecnico}</p>
                  <p><strong>Calificación Cliente:</strong> {evaluacion.calificacion_cliente}</p>
                  <p><strong>Comentarios Cliente:</strong> {evaluacion.comentarios_cliente}</p>
                </div>
                <div style={{display: 'flex', flexDirection: 'column', alignItems: 'center', gap: '4px', minWidth: '20px'}}>
                  {/* Palomita verde para evaluación interna */}
                  <span 
                    className="evaluated-tag" 
                    title="Evaluación interna realizada" 
                    style={{ fontSize: '18px', color: 'green' }}
                  >
                    ✅
                  </span>
                  {/* Palomita verde o X roja para feedback */}
                  {tieneFeedback ? (
                    <span 
                      className="client-feedback-tag" 
                      title="Opinión del cliente presente" 
                      style={{ color: 'green', fontSize: '18px' }}
                    >
                      ✅
                    </span>
                  ) : (
                    <span 
                      className="client-feedback-missing" 
                      title="Sin opinión del cliente" 
                      style={{ color: 'red', fontWeight: 'bold', fontSize: '18px' }}
                    >
                      ❌
                    </span>
                  )}
                </div>
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
};

export default RequestViewer;
