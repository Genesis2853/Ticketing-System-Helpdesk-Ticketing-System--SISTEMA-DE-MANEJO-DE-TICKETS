import React, { useEffect, useState } from 'react';
import { MapContainer, TileLayer, Marker, Popup } from 'react-leaflet';
import L from 'leaflet';
import 'leaflet/dist/leaflet.css';

// Configuración de los iconos de Leaflet
delete L.Icon.Default.prototype._getIconUrl;
L.Icon.Default.mergeOptions({
  iconRetinaUrl: require('leaflet/dist/images/marker-icon-2x.png'),
  iconUrl: require('leaflet/dist/images/marker-icon.png'),
  shadowUrl: require('leaflet/dist/images/marker-shadow.png'),
});

const TecnicosMap = ({ user }) => {
  const [positions, setPositions] = useState([]);
  const [loading, setLoading] = useState(true);

  // Enviar la ubicación al backend
  const enviarUbicacion = (lat, lng) => {
    const token = localStorage.getItem("token");
    fetch(`${process.env.REACT_APP_API_URL_MAPATECNICO}/api/ubicacion`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${token}`,
      },
      body: JSON.stringify({
        codigo_trabajador: user.codigo_trabajador,
        latitud: lat,
        longitud: lng,
      }),
    })
    .then(res => {
      if (!res.ok) {
        throw new Error('Error al enviar la ubicación: ' + res.statusText);
      }
      return res.json();
    })
    .then(data => {
      console.log('Ubicación enviada correctamente:', data);
    })
    .catch(error => {
      console.error('Error enviando ubicación:', error);
    });
  };

  // Obtener ubicación del navegador y enviarla
  useEffect(() => {
    if (!user || !user.codigo_trabajador) return;

    if (navigator.geolocation) {
      navigator.geolocation.getCurrentPosition(
        position => {
          const { latitude, longitude } = position.coords;
          enviarUbicacion(latitude, longitude);
        },
        error => {
          console.warn('Error obteniendo ubicación:', error);
        }
      );
    }
  }, [user]);

  // Obtener ubicaciones de técnicos para el mapa
  useEffect(() => {
    const token = localStorage.getItem("token");

    fetch(`${process.env.REACT_APP_API_URL}/api/ubicacion`, {
      headers: { Authorization: `Bearer ${token}` },
    })
    .then(res => {
      if (!res.ok) {
        throw new Error('Error al obtener ubicaciones: ' + res.statusText);
      }
      return res.json();
    })
    .then(data => {
      setPositions(data); // data debe ser array con {codigo_trabajador, nombre, latitud, longitud}
      setLoading(false);
    })
    .catch(error => {
      console.error('Error fetching data:', error);
      setLoading(false); // Asegúrate de que loading se establezca en false incluso si hay un error
    });
  }, []);

  if (loading) return <div>Cargando mapa...</div>;

  const center = positions.length > 0
    ? [positions[0].latitud, positions[0].longitud]
    : [0, 0];

  return (
    <MapContainer center={center} zoom={13} style={{ height: "100vh", width: "100%" }}>
      <TileLayer
        url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
        attribution="&copy; OpenStreetMap contributors"
      />
      {positions.map(({ codigo_trabajador, nombre, latitud, longitud }) => (
        <Marker key={codigo_trabajador} position={[latitud, longitud]}>
          <Popup>
            Técnico: {nombre} <br />
            Código: {codigo_trabajador}
          </Popup>
        </Marker>
      ))}
    </MapContainer>
  );
};

export default TecnicosMap;
